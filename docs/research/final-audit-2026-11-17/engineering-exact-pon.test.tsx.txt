import { afterEach, beforeEach, describe, expect, it, vi } from 'vitest';
import { useState } from 'react';
import { cleanup, fireEvent, render, screen, within } from '@testing-library/react';
import { ReviewContext } from '../src/hooks/use-review-session';
import { OrderQueueView } from '../src/components/OrderQueueView';
import type { QueueOrder } from '../src/lib/orderQueue';
import { attributeExport, newRx } from './iguanaFixtures';
import { daysAgoMidnight } from '../src/lib/iguana/config';

function Harness() {
  const [orders, setOrders] = useState<QueueOrder[]>([]);
  return <ReviewContext.Provider value={{ ready: true, orders, setOrders, exclusions: [], policyRevision: 0, setExclusions: () => {} }}><OrderQueueView /></ReviewContext.Provider>;
}
const json = (value: unknown) => new Response(JSON.stringify(value), { headers: { 'content-type': 'application/json' } });
beforeEach(() => { localStorage.clear(); });
afterEach(() => { cleanup(); vi.unstubAllGlobals(); });

describe('detected PON lookup uses exact PON identity', () => {
  it('still offers the 7-day search when only a longer suffix sibling is loaded', async () => {
    // TEST-PON-100:1 is a substring of TEST-PON-100:10. They are different orders.
    const detected = 'TEST-PON-100:1';
    const queries = [attributeExport([{ payload: newRx({ id: 'test-msg-10', pon: 'TEST-PON-100:10' }) }]), '<export/>', '<export/>'];
    const fetcher = vi.fn().mockImplementation(async (url: string) => url.endsWith('/detect')
      ? json({ ok: true, pon: detected, token: 'opaque', expiresAt: Date.now() + 300000 })
      : json({ ok: true, status: 200, body: queries.shift() }));
    vi.stubGlobal('fetch', fetcher);
    render(<Harness />);
    fireEvent.click(screen.getByRole('button', { name: 'Load recent E-Rx' }));
    await screen.findByText('Search complete: 1 logs · 1 added. Choose a matching E‑Rx below.');
    fireEvent.click(screen.getByRole('button', { name: 'Detect open E-Rx' }));
    await screen.findByText('Search complete: 0 logs · 0 added. Choose a matching E‑Rx below.');
    expect(JSON.parse(fetcher.mock.calls[2][1].body).filter).toBe(detected);
    // Current b853b88: the substring sibling suppresses this offer.
    expect(await screen.findByText('No matching orders are loaded for the current time window.')).toBeTruthy();
    // The sibling stays visible for the technician; nothing is hidden or auto-selected.
    expect(within(screen.getByRole('region', { name: 'Orders' })).getByText('TEST-PON-100:10')).toBeTruthy();
    fireEvent.click(screen.getByRole('button', { name: 'Search the last 7 days' }));
    await vi.waitFor(() => expect(fetcher).toHaveBeenCalledTimes(4));
    expect(JSON.parse(fetcher.mock.calls[3][1].body)).toMatchObject({ filter: detected, after: daysAgoMidnight(7), before: '' });
    // Still no exact order after the wider search: the empty state remains truthful.
    expect(await screen.findByText('No matching orders are loaded for the current time window.')).toBeTruthy();
  });

  it('keeps substring matching for a typed patient/drug/facility search', async () => {
    const fetcher = vi.fn().mockImplementation(async () => json({ ok: true, status: 200, body: attributeExport([{ payload: newRx({ pon: 'TEST-PON-200' }) }]) }));
    vi.stubGlobal('fetch', fetcher);
    render(<Harness />);
    const search = screen.getByLabelText('PON, patient reference, facility or drug');
    fireEvent.change(search, { target: { value: 'TEST-PON-2' } });
    fireEvent.submit(search.closest('form')!);
    await screen.findByText('Search complete: 1 logs · 1 added. Choose a matching E‑Rx below.');
    expect(screen.queryByText('No matching orders are loaded for the current time window.')).toBeNull();
  });
});
