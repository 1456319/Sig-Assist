import { describe, expect, it } from 'vitest';
import { translateClinicalSig } from '../src/lib/clinical/clinicalEngine';
import { translateFreeTextSig } from '../src/lib/sigEngine';

const clinical = (drugName: string, rawProse: string) =>
  translateClinicalSig({ id: 'H', pon: 'H', drugName, rawProse, sourceFormat: 'manual_text' } as Parameters<typeof translateClinicalSig>[0]);
const LISPRO = 'INSULIN LISPRO 100 UNIT/ML INJECTION';

describe('Nov 18 synthetic holdout', () => {
  it('D1: "N times daily" keeps its count in the clinical translator', () => {
    expect(clinical('METFORMIN 500 MG TABLET', 'Take 1 tablet by mouth three times daily.').primarySig).toBe('1T PO TID');
    expect(clinical('METFORMIN 500 MG TABLET', 'Take 1 tablet by mouth 2 times daily.').primarySig).toBe('1T PO BID');
    expect(clinical('METFORMIN 500 MG TABLET', 'Take 1 tablet by mouth four times daily.').primarySig).toBe('1T PO QID');
  });

  it('D1: a scale given three times daily is not shortened to once daily', () => {
    const r = clinical(LISPRO, 'Inject per sliding scale subcutaneously three times daily: 150-199 2 units, 200-249 4 units.');
    expect(r.primarySig).toBe('CBS TID SS 150-199=2U;200-249=4U SQ');
  });

  it('D2: "Hold if blood sugar less than 70" holds the low band, not the 200-249 dose', () => {
    const r = clinical(LISPRO, 'Inject per sliding scale subcutaneously three times daily: 150-199 2 units, 200-249 4 units. Hold if blood sugar less than 70 and call MD.');
    expect(r.primarySig).toContain('200-249=4U SQ');
    expect(r.primarySig).not.toContain('4U&HOLD');
    expect(r.primarySig).toMatch(/<70=(?:CALL MD&HOLD|HOLD&CALL MD);/);
  });

  it('D3: classic keeps a second dose clause instead of dropping or replacing it', () => {
    expect(translateFreeTextSig('Take 1 tablet by mouth in the morning and take 1 tablet by mouth at noon.', { drug: 'FUROSEMIDE 20 MG TABLET' }).sig)
      .toBe('1T PO QAM. TAKE 1 TABLET BY MOUTH AT NOON');
    expect(translateFreeTextSig('Give 1 capsule by mouth in the morning and give 2 capsules by mouth at bedtime.', { drug: 'GABAPENTIN 100 MG CAPSULE' }).sig)
      .toBe('1C PO QAM. GIVE 2 CAPSULES BY MOUTH AT BEDTIME');
  });

  it('D3: classic keeps monitoring after named meals (both meals retained)', () => {
    const r = translateFreeTextSig('Take 1 tablet by mouth with breakfast and dinner. Check blood sugar before breakfast.', { drug: 'METFORMIN 500 MG TABLET' });
    expect(r.sig).toBe('1T PO WITH BREAKFAST AND DINNER. CHECK BLOOD SUGAR BEFORE BREAKFAST');
    expect(r.order.issues.map(i => i.code)).toContain('retained-supplemental-instructions');
  });

  it('control: multi-site treatment keeps each site schedule', () => {
    expect(clinical('ZINC OXIDE 20% PASTE', 'Cleanse right heel with normal saline, apply zinc oxide daily for 14 days. Cleanse left heel with normal saline, apply zinc oxide twice daily.').primarySig)
      .toBe('TO RT HEEL:CLEANSE W/NORMAL SALINE,THEN APPLY ZINC OXIDE QD X14D,TO LT HEEL:CLEANSE W/NORMAL SALINE,THEN APPLY ZINC OXIDE BID');
  });

  it('control: basal dose plus separately scheduled scale is retained and flagged, not silently merged', () => {
    const r = clinical('INSULIN GLARGINE 100 UNIT/ML INJECTION', 'Inject 12 units subcutaneously at bedtime. Also inject per sliding scale before meals: 150-199 2 units, 200-249 4 units.');
    expect(r.primarySig).toContain('12 UNITS');
    expect(r.abnormalities.map(a => a.id)).toContain('incomplete_sliding_scale');
  });
});
